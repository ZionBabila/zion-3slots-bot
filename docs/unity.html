🎮 MonoBehaviour — The Core
Every Unity script inherits from this

In Unity, every script is a Component attached to a GameObject. It must inherit from MonoBehaviour.

using UnityEngine;

public class PlayerController : MonoBehaviour
{
    // Fields show up in the Inspector!
    public float speed = 5f;
    public int   health = 100;

    void Start()
    {
        // Runs once when the scene starts
        Debug.Log("Player ready!");
    }

    void Update()
    {
        // Runs every frame (~60x per second)
    }
}
🎯
Key rule: Do NOT write a constructor in Unity MonoBehaviour scripts. Unity manages object creation — use Awake() or Start() instead.

⏱️ Unity Lifecycle
Order of execution matters!

Unity calls these methods automatically — you never call them yourself.

1

Awake()

Called first, even if script is disabled. Use for getting component references.

2

OnEnable()

Called when the script/object is enabled.

3

Start()

Called once before the first frame. Script must be enabled. Use for initialisation.

4

Update()

Called every frame. Main loop for gameplay logic, input.

5

FixedUpdate()

Fixed timestep (default 0.02s). Use for physics calculations.

6

LateUpdate()

After all Updates. Good for camera following.

7

OnDisable() / OnDestroy()

Cleanup — unsubscribe events, release resources.

public class Enemy : MonoBehaviour
{
    Rigidbody rb;

    void Awake()
    {
        // Get component ref — even if disabled
        rb = GetComponent<Rigidbody>();
    }

    void Start()
    {
        // Setup after all Awakes complete
        rb.mass = 2f;
    }

    void FixedUpdate()
    {
        // Physics — always use FixedUpdate for Rigidbody
        rb.AddForce(Vector3.forward * 10f);
    }
}
💡
Awake vs Start: Use Awake to get component references, use Start to use them — because all Awakes run before any Start.

🏃 Transform & Movement
Moving, rotating, scaling GameObjects
void Update()
{
    // Move along Z axis
    transform.Translate(Vector3.forward * speed * Time.deltaTime);

    // Set position directly
    transform.position = new Vector3(1f, 0f, 0f);

    // Move toward target
    transform.position = Vector3.MoveTowards(
        transform.position,
        target.position,
        speed * Time.deltaTime
    );
}
🎯
Always multiply movement by Time.deltaTime to make it frame-rate independent.
void Update()
{
    // Rotate around Y axis
    transform.Rotate(Vector3.up * rotSpeed * Time.deltaTime);

    // Look at a target
    transform.LookAt(target.position);

    // Set rotation directly
    transform.rotation = Quaternion.Euler(0, 90, 0);
}
void Update()
{
    // WASD / Arrow Keys (returns -1, 0, or 1)
    float h = Input.GetAxis("Horizontal");
    float v = Input.GetAxis("Vertical");

    Vector3 dir = new Vector3(h, 0, v);
    transform.Translate(dir * speed * Time.deltaTime);

    // Button press
    if (Input.GetKeyDown(KeyCode.Space))
        Jump();

    // Mouse button
    if (Input.GetMouseButtonDown(0))  // 0=left
        Shoot();
}

💥 Collisions & Triggers
Detecting when objects touch

Physical collisions — objects can't pass through each other.

// Requires: Collider + Rigidbody on at least one
void OnCollisionEnter(Collision col)
{
    Debug.Log("Hit: " + col.gameObject.name);
}

void OnCollisionStay(Collision col)
{
    // Runs every frame while touching
}

void OnCollisionExit(Collision col)
{
    // Ran when objects separate
}

Triggers — objects pass through, but you detect the overlap. Check "Is Trigger" on the Collider.

// Requires: Collider with Is Trigger = true
void OnTriggerEnter(Collider other)
{
    if (other.CompareTag("Coin"))
    {
        score++;
        Destroy(other.gameObject);
    }
}

void OnTriggerExit(Collider other)
{
    // Left the zone
}

Triggers are perfect for: coins, checkpoints, damage zones, item pickup areas.

🧩 Prefabs & Instantiate
Spawning objects at runtime

A Prefab is a saved GameObject template. Instantiate() creates a copy of it at runtime — essential for bullets, enemies, coins, etc.

public class Shooter : MonoBehaviour
{
    // Drag your Bullet prefab here in Inspector
    public GameObject bulletPrefab;
    public Transform  firePoint;
    public float      bulletSpeed = 20f;

    void Update()
    {
        if (Input.GetKeyDown(KeyCode.Space))
            Shoot();
    }

    void Shoot()
    {
        // Create bullet at fire point
        GameObject bullet = Instantiate(
            bulletPrefab,
            firePoint.position,
            firePoint.rotation
        );

        // Give it velocity
        Rigidbody rb = bullet.GetComponent<Rigidbody>();
        rb.linearVelocity = firePoint.forward * bulletSpeed;

        // Destroy after 3 seconds
        Destroy(bullet, 3f);
    }
}
💡
Always Destroy spawned objects when no longer needed — otherwise memory fills up.

Common Instantiate patterns

// Simple spawn at position
Instantiate(prefab, pos, Quaternion.identity);

// Spawn as child of another object
Instantiate(prefab, pos, rot, parentTransform);

// Spawn and get the script on it
Enemy e = Instantiate(enemyPrefab, pos, rot)
              .GetComponent<Enemy>();
e.health = 50;

📊 Data & ScriptableObjects
Sharing data between scripts cleanly

A static variable belongs to the class — shared across all instances.

public class GameManager : MonoBehaviour
{
    // Singleton pattern — one GameManager
    public static GameManager Instance;

    public int   score  = 0;
    public int   lives  = 3;
    public bool  gameOver = false;

    void Awake()
    {
        if (Instance == null) Instance = this;
        else Destroy(gameObject);
    }
}

// Access from any other script:
GameManager.Instance.score += 10;

A ScriptableObject is a data container asset — great for weapon stats, character configs, etc.

// Create this as an asset in Unity
[CreateAssetMenu(menuName = "Game/WeaponData")]
public class WeaponData : ScriptableObject
{
    public string weaponName;
    public float  damage;
    public float  fireRate;
    public int    clipSize;
}

// Use in a script:
public class Gun : MonoBehaviour
{
    public WeaponData data; // drag asset here

    void Shoot()
    {
        ApplyDamage(data.damage);
    }
}
// Save data between sessions
PlayerPrefs.SetInt("HighScore", 1500);
PlayerPrefs.SetFloat("Volume", 0.8f);
PlayerPrefs.SetString("PlayerName", "Zion");
PlayerPrefs.Save();

// Load
int    hi   = PlayerPrefs.GetInt("HighScore", 0);
float  vol  = PlayerPrefs.GetFloat("Volume", 1f);
string name = PlayerPrefs.GetString("PlayerName", "");
⚠️
PlayerPrefs is fine for simple saves. For complex game saves use JSON serialization or a proper save system.

⏳ Coroutines
Timed events without blocking

Coroutines let you pause execution and resume after a delay or condition — without freezing the game.

public class SpawnManager : MonoBehaviour
{
    public GameObject enemyPrefab;

    void Start()
    {
        // Start the coroutine
        StartCoroutine(SpawnLoop());
    }

    // Must return IEnumerator
    IEnumerator SpawnLoop()
    {
        while (true)
        {
            Instantiate(enemyPrefab,
                new Vector3(0,0,0),
                Quaternion.identity);

            // Wait 2 seconds then continue
            yield return new WaitForSeconds(2f);
        }
    }
}

// Other yield options:
yield return null;                    // wait 1 frame
yield return new WaitForSeconds(1f); // wait 1 sec
yield return new WaitUntil(()=> isDone); // wait for condition
💡
Use StopCoroutine() or StopAllCoroutines() to stop them. They also stop if the GameObject is destroyed.
